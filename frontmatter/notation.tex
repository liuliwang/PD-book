\section*{第 2 章 基本概念}

{\small
\begin{tabular}{@{}p{0.22\textwidth}@{}p{0.65\textwidth}@{}}
  $\mathcal{B}$ & 物体所占区域 \\
  $\rho(\mathbf{x})$ & 质量密度 \\
  $\mathbf{x}$ & 物质点在参考构型中的位置 \\
  $\mathbf{u}(\mathbf{x},t)$ & 位移场 \\
  $\mathbf{y}(\mathbf{x},t)=\mathbf{x}+\mathbf{u}(\mathbf{x},t)$ & 变形后位置 \\
  $\boldsymbol{\xi}=\mathbf{x}'-\mathbf{x}$ & 参考构型相对位置矢量 \\
  $\boldsymbol{\eta}=\mathbf{u}(\mathbf{x}',t)-\mathbf{u}(\mathbf{x},t)$ & 相对位移矢量 \\
  $\mathbf{f}(\boldsymbol{\eta},\boldsymbol{\xi})$ & 对力函数（pairwise force function） \\
  $\mathbf{b}(\mathbf{x},t)$ & 体力密度 \\
  $\delta$ & 近场范围（horizon） \\
  $H_{\mathbf{x}}$ & 近场域，$H_{\mathbf{x}}=\{\mathbf{x}'\in\mathcal{B}:|\mathbf{x}'-\mathbf{x}|\le\delta\}$ \\
  $\mathcal{F}_{\mathbf{x}}$ & 族，$\mathcal{F}_{\mathbf{x}}=\{\mathbf{x}'\in H_{\mathbf{x}}:\mathbf{x}'\neq\mathbf{x}\}$ \\
  $dV_{\mathbf{x}'}$ & 参考构型中物质点 $\mathbf{x}'$ 处的体积微元 \\
  $s$ & 伸长率（stretch），$s=(|\boldsymbol{\xi}+\boldsymbol{\eta}|-|\boldsymbol{\xi}|)/|\boldsymbol{\xi}|$ \\
  $\boldsymbol{\eta}_{\parallel}$，$\boldsymbol{\eta}_{\perp}$ & 相对位移的轴向分量与偏斜分量，$\boldsymbol{\eta}=\boldsymbol{\eta}_{\parallel}+\boldsymbol{\eta}_{\perp}$（式\eqref{eq:eta-decomp}） \\
  $\mathbf{R}$ & 刚性旋转张量（用于客观性讨论，式\eqref{eq:rigid-motion}） \\
  $w(\boldsymbol{\eta},\boldsymbol{\xi})$ & 成对势（pairwise potential），$\mathbf{f}=\partial w/\partial\boldsymbol{\eta}$（式\eqref{eq:pairwise-force-potential}） \\
  $W(\mathbf{x},t)$ & 应变能密度（strain energy density），$W=\tfrac{1}{2}\int_{H_{\mathbf{x}}} w\, dV_{\boldsymbol{\xi}}$（式\eqref{eq:strain-energy}；态型形式见式\eqref{eq:state-strain-energy}） \\
  $\mathbf{C}(\boldsymbol{\xi})$ & 微模量张量（micromodulus tensor），线性化对力函数 $\mathbf{f}\approx\mathbf{C}(\boldsymbol{\xi})\boldsymbol{\eta}$（式\eqref{eq:micro-modulus}；态型标量形式 $c$ 见第 6 章） \\
  $s_c$ & 临界伸长率（critical stretch），键保持承载的最大伸长率（式\eqref{eq:critical-stretch}） \\
  $\mu(\boldsymbol{\xi},t)$ & 断键历史变量（damage history variable），$\in\{0,1\}$（式\eqref{eq:history-variable}） \\
  $\varphi(\mathbf{x},t)$ & 损伤变量（damage variable），近场域内已断键的比例（式\eqref{eq:damage-variable}） \\
  $g_0(\boldsymbol{\xi})$ & 单键断裂能密度（单位 J/m$^6$），$g_0=\int_0^{\infty}\mathbf{f}\cdot d\boldsymbol{\eta}=w_0(\boldsymbol{\xi})$（式\eqref{eq:bond-fracture-energy}） \\
  $G_c$ & 断裂韧度（fracture toughness），单位裂纹面能量（J/m$^2$），由 $s_c$ 标定（式\eqref{eq:gc-definition}） \\
  $\mathbf{t}(\mathbf{x})$ & 经典连续介质力学的面力（surface traction），N/m$^2$（式\eqref{eq:force-conversion}） \\
  $\mathbf{b}_{\text{eff}}(\mathbf{x})$ & 等效体力层（equivalent body force layer），PD 中代替面力施加于边界薄层（式\eqref{eq:uniform-body-force}） \\
  $\hat{\mathbf{n}}$ & 物体边界外法向单位矢量 \\
\end{tabular}

\section*{非常规态模型（第 7 章）}

\begin{longtable}{@{}p{4.8cm}>{\raggedright\arraybackslash}p{10.2cm}@{}}
  $\mathbf{N}(\mathbf{x}_j)$ & 应变—位移矩阵，$\boldsymbol{\varepsilon}(\mathbf{x}_j)=\mathbf{N}(\mathbf{x}_j)\mathbf{U}(\mathbf{x}_j)$（离散 NOSB） \\
  $\mathbf{Q}(\mathbf{x}_j)$ & 键矢量—力态映射矩阵，编码 $\boldsymbol{\xi}\to\mathbf{K}^{-1}\boldsymbol{\xi}$（$3\times 6$） \\
  $\mathbf{U}_{\mathrm{glob}}$，$\mathbf{F}_{\mathrm{ext}}$ & 整体节点位移向量、外载荷向量 \\
  $\mathbf{K}_{\mathrm{glob}}$ & NOSB 整体刚度矩阵（对称正定，带宽约 $2\delta$） \\
  $\mathbf{R}(\mathbf{U})$ & 残差向量（不平衡力向量），$\mathbf{F}_{\mathrm{ext}}-\mathbf{F}_{\mathrm{int}}(\mathbf{U})$ \\
  $\mathbf{F}_{\mathrm{int}}$ & 节点内力向量 \\
  $\mathbf{K}_{\mathrm{t}}$ & 切线刚度矩阵，$\partial\mathbf{F}_{\mathrm{int}}/\partial\mathbf{U}$ \\
  $\mathbf{A}$ & 材料切线张量，$\partial\mathbf{P}/\partial\mathbf{F}$（含材料部分与初始应力部分） \\
  $\underline{\mathbf{z}}\langle\boldsymbol{\xi}\rangle$ & 变形的非均匀部分（nonuniform part），$\underline{\mathbf{Y}}\langle\boldsymbol{\xi}\rangle-\mathbf{F}\boldsymbol{\xi}$，零能模式的度量 \\
  $\Pi(\mathbf{y})$ & 弹性体总势能（零能模式稳定性分析） \\
  $e(k)$ & 能量型谱度量（正弦位移场下高波数模式携带的能量） \\
  $W^{s}(\underline{\mathbf{Y}})$ & 稳定化应变能密度，$\frac{1}{2}\int_{H_{\mathbf{x}}}\beta\,\underline{\mathbf{z}}\cdot\underline{\mathbf{z}}\,dV_{\mathbf{x}'}$ \\
  $\beta$，$\gamma$ & 稳定化强度参数（$\beta=\gamma\omega$），$\gamma$ 由式 $\gamma=9\kappa\alpha/\bigl(\pi\int_0^\delta\omega r^4\,dr\bigr)$ 标定 \\
  $\mathbf{x}_{s}$ & 应力点（stress point）位置（应力点法） \\
  $\tilde{\mathbf{F}}\langle\boldsymbol{\xi}\rangle$ & 键关联变形梯度，在子近场域 $H_{\mathbf{x}}^{\boldsymbol{\xi}}$ 上构造 \\
  $\mathbf{F}^{\mathrm{bond}}$ & 键级变形梯度（Breitzman–Dayal 能量平均方法） \\
  $\mathbf{F}^{(1)}$，$\mathbf{F}^{(2)}$ & 高阶变形梯度（变形梯度的梯度，应变梯度类量） \\
  $\boldsymbol{\Xi}^{ai}$，$\mathbf{A}^{aij}$，$V^{aijk}$ & CKP 参考构型相对度量：键矢量、面矢量、四面体体积 \\
  $\boldsymbol{\xi}^{ai}$，$\mathbf{a}^{aij}$，$v^{aijk}$ & CKP 变形构型相对度量：键矢量、面矢量、四面体体积 \\
  $\mathbf{T}_1^{ai}$，$\mathbf{T}_2^{ai}$，$\mathbf{T}_3^{ai}$ & CKP 一/二/三邻域力密度 \\
  $\mathbf{L}(\mathbf{x},t)$ & 非局部速度梯度（半拉格朗日框架，当前构型） \\
  $\varepsilon_{\mathrm{tol}}$ & 非线性迭代收敛容差 \\
  $\mathcal{B}$ & 物体所占区域 \\

\section*{本构模型专题（第 8 章）}

\begin{longtable}{@{}p{4.8cm}>{\raggedright\arraybackslash}p{10.2cm}@{}}
  $\Psi(\mathbf{F})$ & 超弹性应变能密度函数（参考构型单位体积），$\mathbf{P}=\partial\Psi/\partial\mathbf{F}$ \\
  $\mathbf{C}$ & 右 Cauchy--Green 张量，$\mathbf{C}=\mathbf{F}^{\mathrm{T}}\mathbf{F}$ \\
  $\mathbf{E}$ & Green--Lagrange 应变张量，$\mathbf{E}=\tfrac{1}{2}(\mathbf{C}-\mathbf{I})$ \\
  $\mathbf{S}$ & 第二类 Piola--Kirchhoff 应力张量，$\mathbf{S}=2\,\partial\Psi/\partial\mathbf{C}$，$\mathbf{P}=\mathbf{F}\mathbf{S}$ \\
  $I_1,I_2,I_3$ & 右 Cauchy--Green 张量的三个主不变量，$I_3=J^{2}$ \\
  $\lambda,\mu$ & Lam\'e 常数（$\mu$ 亦为剪切模量） \\
  $C_{10},C_{01}$ & Mooney--Rivlin 模型材料常数，$2(C_{10}+C_{01})=\mu$ \\
  $\kappa$ & 体积模量 \\
  $\mathbb{C}$ & 材料切线模量张量（四阶），$\mathbb{C}=\partial\mathbf{P}/\partial\mathbf{F}$ \\
  $G(t)$ & 剪切松弛模量（$G_{0}$ 瞬时值，$G_{\infty}$ 平衡值） \\
  $K(t)$ & 体积松弛模量 \\
  $\mathbf{s}$，$\mathbf{e}$ & 偏应力张量、偏应变张量，$\mathbf{s}=\boldsymbol{\sigma}-\tfrac{1}{3}(\operatorname{tr}\boldsymbol{\sigma})\mathbf{I}$ \\
  $\tau_{R}$ & 松弛时间，$\tau_{R}=\eta/E$ \\
  $E$，$\eta$ & 弹簧模量、阻尼器黏度（Maxwell/Kelvin--Voigt 模型） \\
  $\mathbf{h}^{n}$ & 黏弹性记忆项（历史应变递推向量） \\
  $\boldsymbol{\varepsilon}^{e}$，$\boldsymbol{\varepsilon}^{vp}$ & 弹性应变、黏塑性应变 \\
  $\gamma$ & 黏性参数（Perzyna 流动系数） \\
  $f(\boldsymbol{\sigma},\boldsymbol{q})$ & 屈服函数（$\boldsymbol{q}$ 为内变量） \\
  $\langle\cdot\rangle$ & Macaulay 括号（$\langle x\rangle=x$ 若 $x>0$，否则 $0$） \\
  $\sigma_{e}$，$\sigma_{y}$ & von Mises 等效应力、屈服应力 \\
  $J_{2}$ & 偏应力第二不变量，$J_{2}=\tfrac{1}{2}\mathbf{s}:\mathbf{s}$ \\
  $\varepsilon^{c}$，$\boldsymbol{\varepsilon}^{c}$ & 蠕变应变（标量/张量），$\boldsymbol{\varepsilon}^{c}$ 为偏量 \\
  $B$，$n$ & Norton 幂律材料常数（应力指数 $n$） \\
  $D$ & 损伤变量，$D\in[0,1]$（$D=0$ 完好，$D=1$ 失效） \\
  $\tilde{\sigma}$ & 有效应力，$\tilde{\sigma}=\sigma/(1-D)$ \\
  $A$，$q$，$q_{1}$，$q_{2}$ & 损伤演化材料常数（Kachanov--Rabotnov / Liu--Murakami） \\
  $C_{m}$，$n_{2}$ & Liu--Murakami 蠕变材料常数 \\
  $\sigma_{1}$ & 最大主应力 \\
  $D_{\mathrm{cr}}$ & 临界损伤（材料失效判据，通常取 $0.99$） \\
  $\mathbf{m}^{\alpha}$，$\mathbf{n}^{\alpha}$ & 第 $\alpha$ 滑移系的滑移方向单位矢量、滑移面法向单位矢量 \\
  $\mathbf{P}^{\alpha}$ & 第 $\alpha$ 滑移系的 Schmid 张量，$\mathbf{P}^{\alpha}=\mathbf{m}^{\alpha}\otimes\mathbf{n}^{\alpha}$（对称部分 $\mathbf{P}^{\alpha}_{\mathrm{sym}}$） \\
  $\tau^{\alpha}$ & 第 $\alpha$ 滑移系的分切应力（resolved shear stress），$\tau^{\alpha}=\boldsymbol{\sigma}:\mathbf{P}^{\alpha}_{\mathrm{sym}}$ \\
  $\tau_{c}^{\alpha}$ & 第 $\alpha$ 滑移系的临界分切应力（滑移阻力） \\
  $\dot{\gamma}^{\alpha}$ & 第 $\alpha$ 滑移系的滑移率（剪应变率） \\
  $\mathbf{F}^{\mathrm{e}}$，$\mathbf{F}^{\mathrm{p}}$ & 变形梯度的弹性部分、塑性部分（乘法分解 $\mathbf{F}=\mathbf{F}^{\mathrm{e}}\mathbf{F}^{\mathrm{p}}$） \\
  $\mathbf{L}^{\mathrm{p}}$ & 塑性速度梯度，$\mathbf{L}^{\mathrm{p}}=\dot{\mathbf{F}}^{\mathrm{p}}(\mathbf{F}^{\mathrm{p}})^{-1}=\sum_{\alpha}\dot{\gamma}^{\alpha}\,\mathbf{m}^{\alpha}\otimes\mathbf{n}^{\alpha}$ \\
  $\mathbf{R}_{0}$ & 晶格取向矩阵（晶体坐标系到全局坐标系的旋转） \\
  $\dot{\gamma}_{0}$，$N$ & 幂律滑移率参考值、率敏感指数 \\
  $h_{\alpha\beta}$ & 滑移硬化模量矩阵（自硬化 $h_{\alpha\alpha}$，潜硬化 $h_{\alpha\beta}$，$\alpha\neq\beta$） \\
  $q$ & 潜硬化比（潜硬化与自硬化之比，典型值 $1\sim1.4$） \\
  $g^{\alpha}$ & 第 $\alpha$ 滑移系当前滑移阻力（硬化内变量，$\dot{g}^{\alpha}=\sum_{\beta}h_{\alpha\beta}|\dot{\gamma}^{\beta}|$） \\
\end{longtable}
